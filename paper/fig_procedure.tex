%% steps 1-5, once per design family
\def\procstep#1#2#3#4#5{%
  \node[card] (s#1) at (#2,0) {};
  \node[badge=procfam!85!black] (b#1) at ([xshift=0.29cm,yshift=-0.29cm]s#1.north west) {#1};
  \node[ttl=procfam!70!black] at ([xshift=0.11cm]b#1.east) {#3};
  \node[txt] at ([xshift=0.09cm,yshift=-0.60cm]s#1.north west) {#4};
  \draw[procfam!35, line width=0.4pt] ([xshift=0.09cm,yshift=0.95cm]s#1.south west)
    -- ([xshift=-0.09cm,yshift=0.95cm]s#1.south east);
  \node[txt] at ([xshift=0.09cm,yshift=0.87cm]s#1.south west) {\textcolor{procfam!70!black}{\textit{out:}} #5};}
\procstep{1}{1.194}{Frame}{family, alternatives, characteristics; limit on $q_p$; costs}{decision problem}
\procstep{2}{3.862}{Floor}{batches of consecutive parts of the produced alternatives}{$F$ per characteristic}
\procstep{3}{6.530}{Resolve}{effect-to-scatter ratios, minimum resolvable changes}{what production can confirm}
\procstep{4}{9.198}{Calibrate}{rules on the produced alternatives, with and without the simulation}{intervals for $q_p$}
\procstep{5}{11.866}{Qualify}{per family, held out by relation; bands validated out of sample}%
  {$\delta_\mathrm{d}$, $\delta_\mathrm{s}$ and band $g$ per rule and relation}
\foreach \a/\b in {1/2, 2/3, 3/4, 4/5} \draw[flow] (s\a.east) -- (s\b.west);
